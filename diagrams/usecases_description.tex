\section*{Đặc tả Use Case}

Dưới đây là đặc tả chi tiết cho 12 Use Case của hệ thống People Tracking.

% UC01
\begin{table}[H]
\centering
\begin{tabular}{|l|p{12cm}|}
\hline
\textbf{Usecase code} & UC01 \\ \hline
\textbf{Usecase name} & Tạo tài khoản mới \\ \hline
\textbf{Description} & Admin đăng ký/tạo tài khoản mới và cấp vai trò (Role) cho nhân viên. \\ \hline
\textbf{Actors} & Admin \\ \hline
\textbf{Trigger event} & Admin chọn chức năng tạo tài khoản mới trên giao diện. \\ \hline
\textbf{Pre-conditions} & Admin đã đăng nhập thành công vào hệ thống. \\ \hline
\textbf{Post-condition} & Tài khoản mới được lưu vào CSDL và có thể đăng nhập. \\ \hline
\textbf{Main flow} & 
1. Admin bấm tạo người dùng mới. \newline
2. Hệ thống hiển thị form nhập liệu. \newline
3. Admin nhập thông tin (Email, Mật khẩu, Vai trò) và xác nhận. \newline
4. Hệ thống kiểm tra Email/Username chưa tồn tại. \newline
5. Hệ thống lưu tài khoản mới vào cơ sở dữ liệu. \newline
6. Thông báo tạo tài khoản thành công. \\ \hline
\textbf{Alternative flow} & 
1. Thông tin không hợp lệ hoặc Email đã tồn tại. \newline
2. Hệ thống báo lỗi. \newline
3. Quay lại bước 3 của luồng chính. \\ \hline
\end{tabular}
\end{table}

\textbf{Usecase name} & Thêm camera mới (RTSP) \\ \hline
\textbf{Description} & Thêm camera vào hệ thống bằng đường dẫn RTSP/Webcam và kiểm tra kết nối. \\ \hline
\textbf{Actors} & Admin, Operator \\ \hline
\textbf{Trigger event} & Actor chọn chức năng thêm camera RTSP. \\ \hline
\textbf{Pre-conditions} & Actor đã đăng nhập. \\ \hline
\textbf{Post-condition} & Camera mới được cấu hình và lưu vào hệ thống. \\ \hline

1. Actor nhập thông tin camera và đường dẫn RTSP. \newline
2. Actor bấm "Check Connection". \newline
3. Hệ thống ping RTSP bằng OpenCV và trả về thông số (Resolution, FPS). \newline
4. Actor xác nhận lưu camera. \newline
5. Hệ thống lưu thông tin vào CSDL. \\ \hline
\textbf{Alternative flow} & 
1. Luồng RTSP không khả dụng. \newline
2. Hệ thống thông báo lỗi kết nối. \newline
3. Quay lại bước 1 luồng chính. \\ \hline

\textbf{Actors} & Admin, Operator, Viewer \\ \hline
\textbf{Trigger event} & Actor mở giao diện Dashboard của Session đang chạy. \\ \hline

3. Engine sau khi xử lý Frame đẩy dữ liệu vào Queue. \newline
4. Server phát (broadcast) dữ liệu cho các Actor đang kết nối. \newline
5. Giao diện render hình ảnh, hộp giới hạn và biểu đồ. \\ \hline
\textbf{Alternative flow} & 
1. Mất mạng hoặc Server tạm thời ngắt kết nối. \newline
2. Giao diện hiển thị trạng thái "Reconnecting...". \\ \hline
\end{tabular}

\begin{table}[H]

\begin{tabular}{|l|p{12cm}|}
\hline
\textbf{Usecase code} & UC11 \\ \hline
\textbf{Usecase name} & Tìm kiếm người / Lịch sử di chuyển \\ \hline
\textbf{Description} & Tra cứu lộ trình của một người dựa vào ID hoặc ảnh khuôn mặt. \\ \hline
\textbf{Actors} & Admin, Operator, Viewer \\ \hline
\textbf{Trigger event} & Actor tìm kiếm một ID hoặc ảnh đặc trưng. \\ \hline
\textbf{Pre-conditions} & Đã có dữ liệu Tracking lưu trong CSDL pgvector. \\ \hline
\textbf{Post-condition} & Hiển thị danh sách thời điểm và camera xuất hiện đối tượng. \\ \hline
\textbf{Main flow} & 
1. Actor nhập Global ID hoặc tải lên hình ảnh đối tượng. \newline
2. Hệ thống trích xuất vector đặc trưng (nếu đầu vào là ảnh). \newline
3. Hệ thống thực hiện K-NN Search qua pgvector để tìm các Track tương đồng. \newline
4. Hệ thống trả về dữ liệu vị trí và khoảng thời gian. \newline
5. Giao diện vẽ lộ trình di chuyển qua từng camera. \\ \hline
\textbf{Alternative flow} & 
1. Không tìm thấy đối tượng nào có độ tương đồng đủ cao. \newline
2. Hệ thống thông báo "Không có dữ liệu phù hợp". \\ \hline
\end{tabular}
\end{table}
